# Medición de velocidad (para P1)
python3 -c "
import hashlib, time
salt = bytes.fromhex('73346c372d3034')
pw = 'testpassword'
N = 20
t0 = time.perf_counter()
for _ in range(N):
    hashlib.pbkdf2_hmac('sha256', pw.encode(), salt, 200000)
t = time.perf_counter() - t0
print(f'pbkdf2-200k: {N/t:.1f} intentos/s')
t0 = time.perf_counter()
for _ in range(100000):
    h = hashlib.sha256(); h.update(salt + pw.encode()); h.hexdigest()
t = time.perf_counter() - t0
print(f'sha256: {100000/t:.0f} intentos/s')
t0 = time.perf_counter()
for _ in range(100000):
    h = hashlib.md5(); h.update(salt + pw.encode()); h.hexdigest()
t = time.perf_counter() - t0
print(f'md5: {100000/t:.0f} intentos/s')
"
# Resultado: pbkdf2-200k: ~26 intentos/s | sha256: ~1.3M intentos/s | md5: ~1.2M intentos/s
